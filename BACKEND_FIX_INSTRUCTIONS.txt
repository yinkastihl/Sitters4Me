    case 'save_tip':
        $job_id     = (int)($body['job_id']     ?? 0);

        // Verify job exists and belongs to this parent

        // Save tip to jobs table
        run("UPDATE jobs SET tip_amount=? WHERE id=?", [$tip_amount, $job_id]);

        // Also try to create a payments record for the tip
        try {
            run("INSERT INTO payments (job_id, parent_id, sitter_id, amount, payment_type, status, created_at)

        } catch (Exception $e) {
            // payments table may not have all columns — non-critical
        }
